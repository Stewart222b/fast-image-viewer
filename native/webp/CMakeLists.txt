cmake_minimum_required(VERSION 3.16)
project(FastImageViewerWebp C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The embedded codec requires Windows x64.")
endif()
foreach(feature ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
  set(WEBP_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WEBP_USE_THREAD OFF CACHE BOOL "" FORCE)
add_subdirectory(${LIBWEBP_SOURCE} libwebp EXCLUDE_FROM_ALL)
add_library(FastImageViewerWebp SHARED decoder.c ${LIBWEBP_SOURCE}/src/demux/demux.c)
target_include_directories(FastImageViewerWebp PRIVATE ${LIBWEBP_SOURCE} ${LIBWEBP_SOURCE}/src)
target_link_libraries(FastImageViewerWebp PRIVATE webpdecoder)
if(MINGW)
  target_link_options(FastImageViewerWebp PRIVATE -static-libgcc -Wl,--strip-all -Wl,--no-insert-timestamp)
endif()
set_target_properties(FastImageViewerWebp PROPERTIES PREFIX "")
option(FIV_WEBP_BUILD_FIXTURES "Build the optional test fixture generator" OFF)
if(FIV_WEBP_BUILD_FIXTURES)
  add_executable(WebpFixtures ../../tests/create-webp-fixtures.c)
  target_link_libraries(WebpFixtures PRIVATE webp)
endif()
